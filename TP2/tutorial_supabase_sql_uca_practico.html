Crear tabla estudiantes

Pegar el SQL y presionar Run.

crear_tabla_estudiantes.sql
create table if not exists public.estudiantes (
  id bigint generated by default as identity primary key,
  nombre text not null,
  apellido text not null,
  email text unique,
  curso text,
  edad integer,
  activo boolean default true,
  creado_en timestamptz default now()
);
Al presionar Run, Supabase ejecuta la sentencia. Si no hay error, la tabla queda creada.

Consultas de ejemplo

Filtrar, contar y agrupar.

consultas_select.sql
-- 1. Ver estudiantes activos
select *
from public.estudiantes
where activo = true;

-- 2. Buscar estudiantes de un curso
select nombre, apellido, curso
from public.estudiantes
where curso = '2A';

-- 3. Contar estudiantes por curso
select curso, count(*) as cantidad
from public.estudiantes
group by curso
order by cantidad desc;

-- 4. Buscar por parte del apellido
select *
from public.estudiantes
where apellido ilike '%perez%';
Estas consultas no modifican datos: solo muestran información.

ABM: alta, modificación y baja

Ejecutar de a un bloque y observar los resultados.

02_abm_productos.sql
-- ALTA: cargar registros
insert into public.productos_practico
(nombre, categoria, precio, stock, activo)
values
('Mouse inalámbrico', 'Periféricos', 8500.00, 12, true),
('Teclado mecánico', 'Periféricos', 24500.00, 7, true),
('Monitor 24 pulgadas', 'Pantallas', 110000.00, 4, true),
('Cable HDMI', 'Accesorios', 4200.00, 20, true);

-- MODIFICACIÓN: cambiar precio y stock de un producto
update public.productos_practico
set precio = 26000.00, stock = 10
where nombre = 'Teclado mecánico';

-- BAJA lógica: no borra el registro, lo deja inactivo
update public.productos_practico
set activo = false
where nombre = 'Cable HDMI';

-- BAJA física: eliminar un registro puntual
delete from public.productos_practico
where nombre = 'Mouse inalámbrico';
Consigna: explicar brevemente la diferencia entre baja lógica y baja física.

Consultas solicitadas

Resolver y capturar el resultado de cada consulta.

03_select_practico.sql
-- 1. Listar todos los productos
select *
from public.productos_practico
order by id;

-- 2. Mostrar solo productos activos
select nombre, categoria, precio, stock
from public.productos_practico
where activo = true;

-- 3. Buscar productos con precio mayor a 20000
select nombre, precio
from public.productos_practico
where precio > 20000
order by precio desc;

-- 4. Contar productos por categoría
select categoria, count(*) as cantidad
from public.productos_practico
group by categoria
order by cantidad desc;

-- 5. Calcular valor total del stock por producto
select nombre, precio, stock, precio * stock as valor_total_stock
from public.productos_practico
where activo = true;
